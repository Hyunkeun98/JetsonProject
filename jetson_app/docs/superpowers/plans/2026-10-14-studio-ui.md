# jetson-studio Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** `jetson-studio` 하나를 실행하면 브라우저 화면에서 설정(태그 추천 포함) → 정상 데이터 수집 → 학습 → 감시를 끝낼 수 있게 한다. Windows는 exe, Jetson(Linux)은 `uv run`으로 같은 화면을 쓴다.

**Architecture:** 기존 `build_pipeline`/`CalibrationManager`/`deviation`을 그대로 쓰는 얇은 제어 계층(`studio/controller.py`)과, 그 위의 JSON API 서버(`studio/server.py`), 단일 HTML 화면(`web/studio.html`)을 추가한다. 기존 `web_server.py`(읽기 전용)와 CLI는 바꾸지 않는다. torch 없이 import 되도록 기존 모듈의 torch 의존 import를 지연시킨다(Task 1).

**Tech Stack:** Python 3.8, 표준 라이브러리 `http.server`, paho-mqtt<2, PyYAML, pytest, PyInstaller(<6.10, Python 3.8 지원 버전), uv.

**Spec:** `docs/superpowers/specs/2026-10-14-studio-ui-design.md`

## Global Constraints

- Python 3.8 호환: 런타임에 `X | Y`, `dict[str, int]` 같은 문법을 쓰지 않는다(모든 파일 첫 줄 `from __future__ import annotations`, 타입 힌트에만 사용). `str.removeprefix` 등 3.9+ API 금지.
- 패키지/실행은 uv: `uv run pytest`, `uv run jetson-studio`.
- 웹 서버 기본 바인딩은 `127.0.0.1`. 루프백이 아닌 주소는 `--token`이 있을 때만 조작(POST) API를 켠다. 토큰이 없으면 POST는 403.
- 모든 POST는 토큰(설정된 경우, 헤더 `X-Studio-Token`)과 `Host`/`Origin`을 확인한다. 요청 본문 최대 64KB(초과 413).
- 파일 쓰기는 작업 폴더(`workspace/`) 안으로 한정한다. 사용자가 지정한 데이터 폴더는 읽기만 한다.
- v1은 `deviation` 전용. 설정에 GRU(`tags`/`groups`)가 있으면 화면에서 시작을 거부한다(CLI는 그대로).
- 테스트 데이터(`JSONData/`, `models/`, `experiments/`, `workspace/`, PNG)는 커밋하지 않는다(단, 이미 올라간 run3/run4 원본 제외). 새 데이터는 올리기 전에 사용자에게 묻는다.
- 커밋 메시지는 `feat:`/`test:`/`docs:` 접두어를 쓰고, 끝에 `Co-Authored-By: Claude Sonnet 5.5 <noreply@anthropic.com>`를 붙인다.
- 기존 430개 테스트는 계속 통과해야 한다.
- 화면(HTML)에는 외부 URL(CDN, 폰트)을 쓰지 않는다. 동적 문자열은 `textContent`로만 넣는다.

```python
from __future__ import annotations

import threading
from dataclasses import dataclass


@dataclass(frozen=True)
class GroupResult:
    score: float
    top_tag: str


class ActiveModelHolder:
    """SnapshotProcessor(매 스텝, 읽기)와 학습 완료 콜백(train 명령 시, 쓰기)이 서로
    다른 스레드에서 접근하는 현재 엔진을 스레드세이프하게 공유한다."""

    def __init__(self) -> None:
        self._lock = threading.Lock()
        self._engine: object | None = None

    def get(self):
        with self._lock:
            return self._engine

    def set(self, engine) -> None:
        with self._lock:
            self._engine = engine
```

```python
import json
import math

import pytest

from jetson_app.studio import wizard
from jetson_app.studio.errors import StudioError
from jetson_app.timeparse import format_epoch_ns, parse_dx1_timestamp

BASE_NS = parse_dx1_timestamp("2026-10-02T00:00:00Z")


def _timestamp(second):
    return format_epoch_ns(BASE_NS + int(second * 1_000_000_000)).replace("+00:00", "+0000")


def _noise(n, scale):
    return scale * (((n * 7919) % 13) - 6) / 6.0


def _write_data(directory, seconds=300):
    """pos는 가끔 크게 움직이고 평소엔 거의 일정(추천 대상), vel은 늘 흔들림(비추천)."""
    directory.mkdir(parents=True, exist_ok=True)
    with (directory / "DX_0.json").open("w", encoding="utf-8") as f:
        for n in range(seconds * 10):
            moving = (n // 600) % 2 == 1 and (n % 600) < 80
            pos = 100.0 * ((n % 600) / 80.0 if moving else (1.0 if (n // 600) % 2 == 1 else 0.0))
            record = {
                "timestamp": _timestamp(n / 10),
                "m:pos": pos + _noise(n, 1e-5),
                "m:vel": 5.0 * math.sin(n / 3.0) + _noise(n, 0.5),
            }
            f.write(json.dumps({"records": [record]}) + "\n")
```

```python
def analyze(data_dir: str) -> dict:
    """정상 데이터 폴더의 숫자 태그를 분석해 화면에 보여 줄 표를 만든다. 폴더는 읽기만 한다."""

                "kind": r.kind,
                "value_range": r.value_range,

def _clean_form(form: dict) -> dict:
    equipment_id = str(form.get("equipment_id", "")).strip()
    if not _EQUIPMENT_ID.match(equipment_id):
        raise StudioError("장비 이름은 영문, 숫자, _, - 만 쓸 수 있습니다(1~40자).")
    host = str(form.get("host", "")).strip()
    if not host:
        raise StudioError("브로커 주소를 입력하세요(예: localhost).")
    if not _HOST.match(host):
        raise StudioError("브로커 주소에는 포트나 공백을 넣을 수 없습니다. 포트는 아래 칸에 따로 적으세요.")
    try:
        port = int(form.get("port"))
    except (TypeError, ValueError):
        raise StudioError("포트는 1~65535 사이의 숫자여야 합니다.") from None
    if not 1 <= port <= 65535:
        raise StudioError("포트는 1~65535 사이의 숫자여야 합니다.")
    topics = [str(t).strip() for t in form.get("subscribe_topics") or [] if str(t).strip()]
    if not topics:
        raise StudioError("구독할 토픽을 하나 이상 입력하세요(예: dx1/AxisData).")
    tags = []
    for tag in form.get("tags") or []:
        tag = str(tag).strip()
        if tag and tag not in tags:
            tags.append(tag)
    if not tags:
        raise StudioError("점수를 낼 태그를 하나 이상 선택하세요.")
    try:
        min_samples = int(form.get("min_samples", DEFAULT_MIN_SAMPLES))
    except (TypeError, ValueError):

    if min_samples < 1:
        raise StudioError("학습에 필요한 수집 개수는 1 이상이어야 합니다.")
    return {
        "equipment_id": equipment_id,
        "host": host,
        "port": port,
        "topics": topics,
        "tags": tags,
        "min_samples": min_samples,
    }


def _config_dict(clean: dict) -> dict:
    used = set()
    signals = {_signal_name(tag, used): tag for tag in clean["tags"]}
    eid = clean["equipment_id"]
    return {
        "equipment_id": eid,
        "mqtt": {
            "subscribe_topics": clean["topics"],
            "publish_topic": f"jetson/{eid}/anomaly",
            "command_topic": f"jetson/{eid}/cmd",
        },
        "deviation": {"signals": signals},
        "resample_interval_ms": 100,
        "max_lateness_ms": 500,
        "window_size": 30,
        "calibration": {"max_duration": "7d", "min_samples": clean["min_samples"]},
    }


def save_config(workspace: Path, form: dict) -> EquipmentConfig:
    """입력을 검사하고 설정 YAML을 만들어 기존 설정 로더로 검증한 뒤에만 작업 폴더에 저장한다."""
    clean = _clean_form(form)
    config_dir = Path(workspace) / "configs"
    config_dir.mkdir(parents=True, exist_ok=True)
    name = f"{clean['equipment_id']}.yaml"
    final = config_dir / name
    tmp = config_dir / f".{name}.tmp"
    tmp.write_text(
        yaml.safe_dump(_config_dict(clean), allow_unicode=True, sort_keys=False), encoding="utf-8"
    )
    try:
        config = load_equipment_config(tmp)
    except ConfigError as e:
        tmp.unlink()
        raise StudioError(f"설정이 올바르지 않습니다: {e}") from e
    tmp.replace(final)
    (Path(workspace) / SETTINGS_FILE).write_text(
        json.dumps({"config": name, "host": clean["host"], "port": clean["port"]}), encoding="utf-8"
    )
    return config


def load_settings(workspace: Path):
    path = Path(workspace) / SETTINGS_FILE
    if not path.exists():
        return None
    try:
        data = json.loads(path.read_text(encoding="utf-8"))
    except ValueError:
        return None
    if not isinstance(data, dict) or not {"config", "host", "port"} <= set(data):
        return None
    return data


def saved_form(workspace: Path) -> dict:
    """저장된 설정을 화면 입력칸 값으로 되돌린다. 아직 저장한 것이 없으면 빈 딕셔너리."""
    settings = load_settings(workspace)
    if settings is None:
        return {}
    try:
        config = load_equipment_config(Path(workspace) / "configs" / settings["config"])
    except ConfigError:
        return {}
    signals = config.deviation.signals if config.deviation else ()
    return {
        "equipment_id": config.equipment_id,
        "host": settings["host"],
        "port": settings["port"],
        "subscribe_topics": list(config.subscribe_topics),
        "tags": [s.tag for s in signals],
        "min_samples": config.calibration.min_samples,
    }
```

```python
새로 계산하는 로직은 없다. 파이프라인 생성은 build_pipeline, 학습은 CalibrationManager와
make_deviation_train_fn, 태그 추천은 deviation_suggest를 그대로 부른다.
"""
from __future__ import annotations

import threading
import time
from enum import Enum
from pathlib import Path

from ..calibration import CalibrationState
from ..config import ConfigError, load_equipment_config
from ..deviation import deviation_artifact_path, load_deviation_artifact, make_deviation_train_fn

from .errors import StudioError

NO_DATA_WARN_SECONDS = 30.0
_EMPTY_SCORES = {"now_ms": None, "bucket_ms": 0, "points": []}


class Stage(str, Enum):
    SETUP = "SETUP"            # 저장된 설정이 없다
    READY = "READY"            # 설정은 있고 아직 브로커에 연결하지 않았다
    COLLECTING = "COLLECTING"  # 연결됨, 정상 데이터를 모으는 중
    TRAINING = "TRAINING"      # 학습 스레드가 도는 중
    MONITORING = "MONITORING"  # 학습 결과로 감시 중


class MqttIO:
    """파이프라인의 MQTT 클라이언트를 백그라운드 스레드로 돌리고 멈춘다(테스트에서는 가짜로 바꾼다)."""

    def connect(self, pipeline, host: str, port: int) -> None:
        pipeline.mqtt_subscriber.connect(host, port)
        pipeline.mqtt_subscriber.client.loop_start()

    def disconnect(self, pipeline) -> None:
        client = pipeline.mqtt_subscriber.client
        client.loop_stop()
        client.disconnect()

    def is_connected(self, pipeline) -> bool:
        return bool(pipeline.mqtt_subscriber.client.is_connected())


class StudioController:
    def __init__(
        self,
        workspace,
        io=None,
        pipeline_factory=build_pipeline,
        clock=time.monotonic,
        run_worker: bool = True,
    ) -> None:
        self._workspace = Path(workspace)
        self._workspace.mkdir(parents=True, exist_ok=True)
        self._calibration_dir = self._workspace / "calibration"
        self._model_dir = self._workspace / "models"
        self._io = io or MqttIO()
        self._factory = pipeline_factory
        self._clock = clock
        self._run_worker = run_worker
        self._lock = threading.RLock()
        self._pipeline = None
        self._config = None
        self._training = False
        self._train_thread = None
        self._error = None
        self._last_count = 0
        self._last_rx = None
        self._started_at = 0.0
        self._load_saved_config()

    # --- 설정 ---------------------------------------------------------------------------

    def _load_saved_config(self) -> None:
        settings = wizard.load_settings(self._workspace)
        if settings is None:
            return
        try:
            self._config = load_equipment_config(self._workspace / "configs" / settings["config"])
        except ConfigError as e:
            self._error = f"저장된 설정을 읽지 못했습니다: {e}"

    def analyze(self, data_dir: str) -> dict:
        return wizard.analyze(data_dir)

    def form(self) -> dict:
        return wizard.saved_form(self._workspace)

    def save_config(self, form: dict) -> None:
        with self._lock:
            if self._pipeline is not None:
                raise StudioError("실행 중에는 설정을 바꿀 수 없습니다. 먼저 중지하세요.")
            self._config = wizard.save_config(self._workspace, form)
            self._error = None

    # --- 실행 ---------------------------------------------------------------------------

    @property
    def pipeline(self):
        return self._pipeline

    def start(self) -> None:
        with self._lock:
            if self._config is None:
                raise StudioError("먼저 설정을 저장하세요.")
            if self._pipeline is not None:
                raise StudioError("이미 실행 중입니다.")
            config = self._config
            if config.tags or config.deviation is None:
                raise StudioError("이 화면은 위치 신호의 기준값 이탈(deviation) 설정만 지원합니다.")
            settings = wizard.load_settings(self._workspace)
            host, port = settings["host"], settings["port"]
            train_fn = make_deviation_train_fn(
                config, deviation_artifact_path(self._model_dir, config.equipment_id)
            )
            pipeline = self._factory(
                config=config,
                calibration_dir=self._calibration_dir,
                model_dir=self._model_dir,
                train_fn=train_fn,
                history_seconds=DEFAULT_HISTORY_SECONDS,
            )
            try:
                self._io.connect(pipeline, host, port)
            except OSError as e:
                raise StudioError(
                    f"브로커({host}:{port})에 연결할 수 없습니다. Mosquitto가 실행 중인지 확인하세요. ({e})"
                ) from e
            if self._run_worker:
                pipeline.snapshotter.start()
            self._pipeline = pipeline
            self._error = None
            self._last_count = pipeline.calibration_manager.sample_count
            self._last_rx = None
            self._started_at = self._clock()

    def stop(self) -> None:
        with self._lock:
            if self._pipeline is None:
                return
            if self._training:
                raise StudioError("학습이 끝난 뒤에 중지할 수 있습니다.")
            pipeline = self._pipeline
            self._pipeline = None
        if self._run_worker:
            pipeline.snapshotter.stop()
        self._io.disconnect(pipeline)

    def feed_record(self, record) -> None:
        """MQTT 스레드가 하는 일(리샘플 후 처리 큐에 넣기)을 그대로 한다. 시험과 재생용."""
        pipeline = self._pipeline
        if pipeline is None:
            raise StudioError("실행 중이 아닙니다.")
        pipeline.snapshotter.submit(pipeline.resampler.add(record))

    def pump(self) -> None:
        """처리 큐를 동기로 비운다(run_worker=False일 때 시험이 쓴다)."""
        if self._pipeline is not None:
            self._pipeline.snapshotter.process_pending()

    # --- 학습 ---------------------------------------------------------------------------

    def train(self) -> None:
        with self._lock:
            pipeline = self._pipeline
            if pipeline is None:
                raise StudioError("먼저 시작해서 정상 데이터를 모으세요.")
            if self._training:
                raise StudioError("이미 학습 중입니다.")
            manager = pipeline.calibration_manager
            if manager.state != CalibrationState.CALIBRATING:
                raise StudioError("이미 학습이 끝났습니다. 다시 학습하려면 '처음부터 다시 수집'을 누르세요.")
            if manager.sample_count < manager.min_samples:
                raise StudioError(
                    f"정상 데이터가 부족합니다: {manager.sample_count} / {manager.min_samples}"
                )
            self._training = True
            self._error = None

        def _run() -> None:
            try:
                manager.handle_train_command()
            except Exception as e:
                with self._lock:
                    self._error = f"학습에 실패했습니다: {e}"
            finally:
                with self._lock:
                    self._training = False

        self._train_thread = threading.Thread(target=_run, daemon=True)
        self._train_thread.start()

    def join_training(self, timeout=None) -> None:
        thread = self._train_thread
        if thread is not None:
            thread.join(timeout)

    def recalibrate(self) -> None:
        with self._lock:
            if self._pipeline is None:
                raise StudioError("실행 중이 아닙니다.")
            if self._training:
                raise StudioError("학습이 끝난 뒤에 다시 수집할 수 있습니다.")
            self._pipeline.calibration_manager.handle_recalibrate_command()
            self._error = None
            self._last_count = 0
            self._last_rx = None
            self._started_at = self._clock()

    # --- 화면용 조회 ----------------------------------------------------------------------

    def _stage_locked(self) -> Stage:
        if self._config is None:
            return Stage.SETUP
        if self._pipeline is None:
            return Stage.READY
        if self._training:
            return Stage.TRAINING
        if self._pipeline.calibration_manager.state == CalibrationState.MONITORING:
            return Stage.MONITORING
        return Stage.COLLECTING

    def _signals(self, config) -> list:
        path = deviation_artifact_path(self._model_dir, config.equipment_id)
        if not path.exists():
            return []
        try:
            artifact = load_deviation_artifact(path)
        except Exception:
            return []
        return [
            {"name": n, "tag": s.tag, "sigma": s.sigma, "limit": s.limit, "move_range": s.move_range}
            for n, s in artifact.signals.items()
        ]

    def state(self) -> dict:
        with self._lock:
            config, pipeline = self._config, self._pipeline
            stage = self._stage_locked()
            info = {
                "stage": stage.value,
                "equipment_id": config.equipment_id if config else None,
                "error": self._error,
                "connected": False,
                "collected": 0,
                "required": config.calibration.min_samples if config else 0,
                "no_data": False,
                "signals": [],
                "threshold": config.alarm.threshold if config else None,
            }
            if pipeline is not None:
                info["connected"] = bool(self._io.is_connected(pipeline))
                count = pipeline.calibration_manager.sample_count
                now = self._clock()
                if count != self._last_count:
                    self._last_count = count
                    self._last_rx = now
                info["collected"] = count
                if stage == Stage.COLLECTING:
                    since = self._last_rx if self._last_rx is not None else self._started_at
                    info["no_data"] = (now - since) > NO_DATA_WARN_SECONDS
            if stage == Stage.MONITORING and config is not None:
                info["signals"] = self._signals(config)
            return info

    def meta(self) -> dict:
        """기존 점수 그래프 화면(dashboard.html)이 읽는 /api/meta와 같은 모양."""
        with self._lock:
            config, pipeline = self._config, self._pipeline
            if config is None or pipeline is None or pipeline.history is None:
                return {
                    "equipment_id": config.equipment_id if config else "",
                    "groups": [],
                    "threshold": config.alarm.threshold if config else 0,
                    "confirm_steps": config.alarm.confirm_steps if config else 1,
                    "interval_ms": config.resample_interval_ms if config else 100,
                    "history_seconds": DEFAULT_HISTORY_SECONDS,
                    "state": self._stage_locked().value,
                }
            return {
                "equipment_id": config.equipment_id,
                "groups": list(pipeline.history.group_names),
                "threshold": config.alarm.threshold,
                "confirm_steps": config.alarm.confirm_steps,
                "interval_ms": config.resample_interval_ms,
                "history_seconds": DEFAULT_HISTORY_SECONDS,
                "state": pipeline.calibration_manager.state.value,
            }

    def scores(self, seconds: float, max_points: int) -> dict:
        pipeline = self._pipeline
        if pipeline is None or pipeline.history is None:
            return dict(_EMPTY_SCORES)
        return pipeline.history.snapshot(seconds, max_points)
```

```python
"""jetson-studio의 화면과 JSON API를 내보내는 서버(표준 라이브러리, 백그라운드 스레드)."""
from __future__ import annotations

import hmac
import json
import threading
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
from pathlib import Path
from urllib.parse import parse_qs, urlparse

from ..dashboard_page import render_dashboard
from .controller import StudioController
from .errors import StudioError

MAX_BODY_BYTES = 64 * 1024
LOOPBACK_HOSTS = ("127.0.0.1", "localhost", "::1")
_STUDIO_PAGE = Path(__file__).resolve().parent.parent / "web" / "studio.html"


class StudioServer:
    def __init__(
        self,
        controller: StudioController,
        host: str = "127.0.0.1",
        port: int = 8080,
        token=None,
        control_enabled: bool = True,
    ) -> None:
        self._controller = controller
        self._token = token or None
        self._control_enabled = control_enabled
        self._loopback = host in LOOPBACK_HOSTS
        self._studio_page = _STUDIO_PAGE.read_bytes()
        self._dashboard_page = render_dashboard({"mode": "live"}).encode("utf-8")
        self._post_routes = {
            "/api/analyze": lambda body: controller.analyze(str(body.get("data_dir", ""))),
            "/api/config": lambda body: controller.save_config(body),
            "/api/start": lambda body: controller.start(),
            "/api/train": lambda body: controller.train(),
            "/api/recalibrate": lambda body: controller.recalibrate(),
            "/api/stop": lambda body: controller.stop(),
        }
        server = self

        class Handler(BaseHTTPRequestHandler):
            def do_GET(self):
                server._get(self)

            def do_POST(self):
                server._post(self)

            def _unsupported(self):
                server._send_json(self, 405, {"error": "지원하지 않는 요청입니다"})

            do_PUT = do_DELETE = do_PATCH = _unsupported

            def log_message(self, fmt, *args):
                pass

        self._httpd = ThreadingHTTPServer((host, port), Handler)
        self._httpd.daemon_threads = True
        self._thread = None

    @property
    def port(self) -> int:
        return self._httpd.server_address[1]

    def start(self) -> None:
        self._thread = threading.Thread(target=self._httpd.serve_forever, daemon=True)
        self._thread.start()

    def stop(self) -> None:
        self._httpd.shutdown()
        self._httpd.server_close()
        if self._thread is not None:
            self._thread.join()

    # --- GET ----------------------------------------------------------------------------

    def _get(self, h) -> None:
        url = urlparse(h.path)
        query = parse_qs(url.query)
        c = self._controller
        if url.path == "/":
            self._send(h, 200, self._studio_page, "text/html; charset=utf-8")
        elif url.path == "/dashboard":
            self._send(h, 200, self._dashboard_page, "text/html; charset=utf-8")
        elif url.path == "/healthz":
            self._send_json(h, 200, {"ok": True})
        elif url.path == "/api/state":
            state = c.state()
            state["control_enabled"] = self._control_enabled
            state["token_required"] = self._token is not None
            self._send_json(h, 200, state)
        elif url.path == "/api/config":
            self._send_json(h, 200, c.form())
        elif url.path == "/api/meta":
            self._send_json(h, 200, c.meta())
        elif url.path == "/api/scores":
            self._send_json(
                h, 200, c.scores(_number(query, "seconds", 300.0), int(_number(query, "max_points", 1200)))
            )
        else:
            self._send(h, 404, b"not found", "text/plain; charset=utf-8")

    # --- POST ---------------------------------------------------------------------------

    def _post(self, h) -> None:
        if not self._control_enabled:
            return self._send_json(
                h, 403, {"error": "조작 기능이 꺼져 있습니다. 다른 PC에서 조작하려면 --token을 지정해 다시 시작하세요."}
            )
        if not self._origin_ok(h) or not self._token_ok(h):
            return self._send_json(h, 403, {"error": "허용되지 않은 요청입니다(토큰 또는 출처 확인 실패)."})
        route = self._post_routes.get(urlparse(h.path).path)
        if route is None:
            return self._send_json(h, 404, {"error": "없는 경로입니다"})
        try:
            length = int(h.headers.get("Content-Length") or 0)
        except ValueError:
            return self._send_json(h, 400, {"error": "Content-Length가 올바르지 않습니다"})
        if length > MAX_BODY_BYTES:
            return self._send_json(h, 413, {"error": "요청이 너무 큽니다"})
        raw = h.rfile.read(length) if length else b""
        try:
            body = json.loads(raw) if raw else {}
        except ValueError:
            return self._send_json(h, 400, {"error": "JSON 형식이 올바르지 않습니다"})
        if not isinstance(body, dict):
            return self._send_json(h, 400, {"error": "JSON 객체를 보내야 합니다"})
        try:
            result = route(body)
        except StudioError as e:
            return self._send_json(h, 400, {"error": str(e)})
        except Exception as e:
            print(f"[studio] 요청 처리 중 오류: {e!r}")
            return self._send_json(h, 500, {"error": "내부 오류가 났습니다. 콘솔 로그를 확인하세요."})
        self._send_json(h, 200, result if isinstance(result, dict) else {"ok": True})

    def _origin_ok(self, h) -> bool:
        host = h.headers.get("Host", "")
        hostname = urlparse("//" + host).hostname
        if self._loopback and hostname not in LOOPBACK_HOSTS:
            return False              # DNS rebinding 방어: 루프백 서버는 루프백 이름으로 온 요청만 받는다
        origin = h.headers.get("Origin")
        if origin is not None and urlparse(origin).netloc != host:
            return False              # 다른 사이트의 페이지에서 보낸 요청
        return True

    def _token_ok(self, h) -> bool:
        if self._token is None:
            return True
        supplied = h.headers.get("X-Studio-Token", "")
        return hmac.compare_digest(supplied.encode("utf-8"), self._token.encode("utf-8"))

    # --- 응답 ---------------------------------------------------------------------------

    def _send_json(self, h, status: int, payload) -> None:
        body = json.dumps(payload, ensure_ascii=False).encode("utf-8")
        self._send(h, status, body, "application/json; charset=utf-8")

    @staticmethod
    def _send(h, status: int, body: bytes, content_type: str) -> None:
        h.send_response(status)
        h.send_header("Content-Type", content_type)
        h.send_header("Content-Length", str(len(body)))
        h.send_header("Cache-Control", "no-store")
        h.send_header("X-Content-Type-Options", "nosniff")
        h.end_headers()
        h.wfile.write(body)


def _number(query: dict, key: str, default: float) -> float:
    try:
        value = float(query[key][0])
    except (KeyError, IndexError, ValueError):
        return default
    if value != value or value in (float("inf"), float("-inf")):
        return default
    return value
```

```python
"""화면이 쓰는 HTTP API만으로 설정 → 수집 → 학습 → 감시 → 충격 알람까지 끝낸다(MQTT는 가짜)."""
import json
import urllib.request
from datetime import datetime, timedelta, timezone
from types import SimpleNamespace

from jetson_app.studio.controller import StudioController
from jetson_app.studio.server import StudioServer


class FakeIO:
    def connect(self, pipeline, host, port):
        pass

    def disconnect(self, pipeline):
        pass

    def is_connected(self, pipeline):
        return True


def _call(server, method, path, body=None):
    data = None if body is None else json.dumps(body).encode("utf-8")
    request = urllib.request.Request(f"http://127.0.0.1:{server.port}{path}", data=data, method=method)
    with urllib.request.urlopen(request) as response:
        return json.loads(response.read().decode("utf-8"))


def _ts(i):
    moment = datetime(2026, 8, 4, tzinfo=timezone.utc) + timedelta(milliseconds=i * 100)
    return moment.strftime("%Y-%m-%dT%H:%M:%S.%f") + "000+0000"


def _noise(i):
    return 0.001 * (((i * 7919) % 13) - 6) / 6.0


def _feed(controller, start, stop, spike_at=None):
    for i in range(start, stop):
        value = _noise(i) + (0.01 if i == spike_at else 0.0)
        payload = json.dumps({"records": [{"timestamp": _ts(i), "pos": value}]}).encode("utf-8")
        controller.pipeline.mqtt_subscriber._handle_message(None, None, SimpleNamespace(payload=payload))
    controller.pump()


def test_whole_workflow_over_http(tmp_path):
    controller = StudioController(tmp_path, io=FakeIO(), run_worker=False)
    server = StudioServer(controller, port=0)
    server.start()
    try:
        assert _call(server, "GET", "/api/state")["stage"] == "SETUP"
        _call(server, "POST", "/api/config", {
            "equipment_id": "demo", "host": "localhost", "port": 1883,
            "subscribe_topics": ["dx1/data"], "tags": ["pos"], "min_samples": 200,
        })
        _call(server, "POST", "/api/start", {})
        _feed(controller, 0, 400)
        assert _call(server, "GET", "/api/state")["collected"] >= 200

        _call(server, "POST", "/api/train", {})
        controller.join_training(timeout=30)
        state = _call(server, "GET", "/api/state")
        assert state["stage"] == "MONITORING" and state["signals"][0]["name"] == "pos"

        _feed(controller, 400, 700, spike_at=500)
        points = _call(server, "GET", "/api/scores?seconds=3600&max_points=5000")["points"]
        assert sum(1 for p in points if p["g"]["pos"][1]) == 1

        _call(server, "POST", "/api/recalibrate", {})
        assert _call(server, "GET", "/api/state")["stage"] == "COLLECTING"
        _call(server, "POST", "/api/stop", {})
        assert _call(server, "GET", "/api/state")["stage"] == "READY"
    finally:
        server.stop()
```

```python
    parser.add_argument("--no-browser", action="store_true", help="브라우저를 자동으로 열지 않는다")
    args = parser.parse_args(argv)

    workspace = Path(args.workspace) if args.workspace else default_workspace()
    enabled = control_enabled(args.web_host, args.token)
    try:
        controller = StudioController(workspace)
        server = StudioServer(
            controller, host=args.web_host, port=args.web_port, token=args.token, control_enabled=enabled
        )
    except OSError as e:
        print(f"error: {args.web_host}:{args.web_port} 또는 작업 폴더를 열 수 없습니다: {e}", file=sys.stderr)
        raise SystemExit(2)

    server.start()
    shown_host = "localhost" if args.web_host in LOOPBACK_HOSTS else args.web_host
    url = f"http://{shown_host}:{server.port}/" + (f"?token={args.token}" if args.token else "")
    print(f"[studio] 화면: {url}")
    print(f"[studio] 작업 폴더: {workspace}")
    if not enabled:
        print(
            "[studio] 경고: 루프백이 아닌 주소인데 --token이 없어 조작 기능을 껐습니다(읽기 전용).",
            file=sys.stderr,
        )
    if not args.no_browser:
        threading.Timer(0.5, webbrowser.open, args=(url,)).start()
    try:
        threading.Event().wait()
    except KeyboardInterrupt:
        print("\n중단됨")
    finally:
        try:
            controller.stop()
        except Exception as e:
            print(f"[studio] 종료 정리 중 오류(무시): {e}", file=sys.stderr)
        server.stop()


if __name__ == "__main__":
    main()
```
